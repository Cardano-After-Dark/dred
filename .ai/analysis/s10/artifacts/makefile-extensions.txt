.PHONY: demo-query-servers
demo-query-servers:
	@echo "🔍 Querying both DRED servers..."
	@cd .ai/analysis/s10/artifacts && bash query-servers.sh

.PHONY: demo-replication
demo-replication:
	@echo "🚀 Starting replication demonstration..."
	@cd .ai/analysis/s10/artifacts && bash replication-demo.sh

.PHONY: demo-replication-auto
demo-replication-auto:
	@echo "🚀 Running automated replication demonstration..."
	@cd .ai/analysis/s10/artifacts && bash replication-demo.sh --auto

.PHONY: demo-create-channel
demo-create-channel:
	@if [ -z "$(CHANNEL)" ]; then \
		echo "Usage: make demo-create-channel CHANNEL=your-channel-name"; \
		exit 1; \
	fi
	@echo "📝 Creating channel '$(CHANNEL)' on both servers..."
	@curl -X POST http://$(US):$(DRED_PORT)/channel/$(CHANNEL) -H "Content-Type: application/json" -d '{"createdAt": '$(shell date +%s000)'}' && echo " ✓ US"
	@curl -X POST http://$(UK):$(DRED_PORT)/channel/$(CHANNEL) -H "Content-Type: application/json" -d '{"createdAt": '$(shell date +%s000)'}' && echo " ✓ UK"

.PHONY: demo-send-message
demo-send-message:
	@if [ -z "$(CHANNEL)" ] || [ -z "$(MESSAGE)" ]; then \
		echo "Usage: make demo-send-message CHANNEL=your-channel MESSAGE='your message' [SERVER=us|uk]"; \
		exit 1; \
	fi
	@if [ "$(SERVER)" = "uk" ]; then \
		echo "📤 Sending message to UK server..."; \
		curl -X POST http://$(UK):$(DRED_PORT)/channel/$(CHANNEL)/message \
			-H "Content-Type: application/json" \
			-d '{"msg":"$(MESSAGE)","type":"demo","ocid":"demo-'$(shell date +%s)'","sender":"makefile-demo"}'; \
	else \
		echo "📤 Sending message to US server..."; \
		curl -X POST http://$(US):$(DRED_PORT)/channel/$(CHANNEL)/message \
			-H "Content-Type: application/json" \
			-d '{"msg":"$(MESSAGE)","type":"demo","ocid":"demo-'$(shell date +%s)'","sender":"makefile-demo"}'; \
	fi
	@echo ""

.PHONY: demo-test-remote-replication  
demo-test-remote-replication:
	@echo "🧪 Running remote replication TypeScript test..."
	@echo "⚠️  This test requires both servers to be operational and in the same neighborhood"
	@pnpm test remote-replication
